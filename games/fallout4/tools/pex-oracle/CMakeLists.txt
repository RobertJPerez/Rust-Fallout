cmake_minimum_required(VERSION 3.20)
project(fo4_pex_oracle LANGUAGES CXX)
if(NOT CHAMPOLLION_SOURCE)
  message(FATAL_ERROR "Pass the pinned CHAMPOLLION_SOURCE checkout")
endif()
file(GLOB PEX_SOURCES "${CHAMPOLLION_SOURCE}/Pex/*.cpp")
add_executable(fo4-pex-oracle main.cpp ${PEX_SOURCES})
target_include_directories(fo4-pex-oracle PRIVATE "${CHAMPOLLION_SOURCE}")
target_compile_features(fo4-pex-oracle PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(fo4-pex-oracle PRIVATE /FIcstring /FItype_traits)
endif()
