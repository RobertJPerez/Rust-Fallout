cmake_minimum_required(VERSION 3.20)
project(fallout_actor_oracle LANGUAGES CXX)
add_executable(actor-oracle main.cpp)
target_compile_features(actor-oracle PRIVATE cxx_std_17)
if(NOT WIN32)
    message(FATAL_ERROR "This offline reader uses Windows read locks and CNG")
endif()
target_compile_definitions(actor-oracle PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(actor-oracle PRIVATE bcrypt)
