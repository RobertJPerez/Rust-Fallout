cmake_minimum_required(VERSION 3.20)
project(fallout_condition_oracle LANGUAGES CXX)
add_executable(condition-oracle main.cpp)

endif()
target_compile_definitions(condition-oracle PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(condition-oracle PRIVATE bcrypt)
