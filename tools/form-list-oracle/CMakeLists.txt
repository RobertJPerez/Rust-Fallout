cmake_minimum_required(VERSION 3.20)
project(fallout_form_list_oracle LANGUAGES CXX)
add_executable(form-list-oracle main.cpp)
target_compile_features(form-list-oracle PRIVATE cxx_std_17)
if(NOT WIN32)
    message(FATAL_ERROR "This offline reader uses Windows source locks and CNG")
endif()
target_compile_definitions(form-list-oracle PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(form-list-oracle PRIVATE bcrypt)
