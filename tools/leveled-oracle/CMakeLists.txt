cmake_minimum_required(VERSION 3.20)
project(fallout_leveled_oracle LANGUAGES CXX)
add_executable(leveled-oracle main.cpp)
target_compile_features(leveled-oracle PRIVATE cxx_std_17)
if(NOT WIN32)
    message(FATAL_ERROR "This offline reader uses Windows source locks and CNG")
endif()
target_compile_definitions(leveled-oracle PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(leveled-oracle PRIVATE bcrypt)
