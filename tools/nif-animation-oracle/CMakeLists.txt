cmake_minimum_required(VERSION 3.20)
project(fallout_nif_animation_oracle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "No upstream test models" FORCE)
if(NOT DEFINED NIFLY_SOURCE_DIR)
    message(FATAL_ERROR "Set NIFLY_SOURCE_DIR to the read-only pinned checkout")
endif()
add_subdirectory("${NIFLY_SOURCE_DIR}" nifly-build)
add_executable(nif-animation-oracle main.cpp)
target_link_libraries(nif-animation-oracle PRIVATE nifly bcrypt)
target_compile_definitions(nif-animation-oracle PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
