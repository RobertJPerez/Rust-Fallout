cmake_minimum_required(VERSION 3.20)
project(fallout_nif_oracle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "No upstream sample assets" FORCE)
if(NOT DEFINED NIFLY_SOURCE_DIR)
    message(FATAL_ERROR "Set NIFLY_SOURCE_DIR to the pinned research checkout")
endif()
add_subdirectory("${NIFLY_SOURCE_DIR}" nifly-build)
add_executable(nif-oracle main.cpp)
target_link_libraries(nif-oracle PRIVATE nifly)
