cmake_minimum_required(VERSION 3.20)
project(fallout_script_oracle LANGUAGES CXX)
add_executable(script-oracle main.cpp)
target_compile_features(script-oracle PRIVATE cxx_std_17)
if(NOT WIN32)
    message(FATAL_ERROR "This offline comparison uses Windows CNG hashing")
endif()
target_compile_definitions(script-oracle PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(script-oracle PRIVATE bcrypt)
