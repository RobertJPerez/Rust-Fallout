cmake_minimum_required(VERSION 3.20)
project(fallout_terrain_oracle LANGUAGES CXX)
add_executable(terrain-oracle main.cpp)
target_compile_features(terrain-oracle PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(terrain-oracle PRIVATE /fp:strict)
endif()
if(NOT WIN32)
    message(FATAL_ERROR "This offline oracle uses Windows CNG for hashing")
endif()
target_compile_definitions(terrain-oracle PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(terrain-oracle PRIVATE bcrypt)
